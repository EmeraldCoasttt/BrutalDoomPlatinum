class BDP_DualPlasmaRifle : BDP_Weapon
{
	action state A_LowSprites(string nosprite, string lowsprite, string medsprite, string highsprite, bool dual = false)
	{
		let psp = player.FindPSprite(OverlayID());
		let weap = BDP_Weapon(FindInventory(invoker.DualWeapon));
		BDP_Weapon target;
		if(dual)
			target = weap;
		else
			target = invoker;
		if(psp)
		{
			if(target.mag < 1 || target.mag < 11 && level.time % 8 < 4)
				psp.sprite = GetSpriteIndex(nosprite);
			else if(target.mag < 11)
				psp.sprite = GetSpriteIndex(lowsprite);
			else if(target.mag < 21)
				psp.sprite = GetSpriteIndex(medsprite);
			else
				psp.sprite = GetSpriteIndex(highsprite);
		}
		return resolvestate(null);
	}
	
	int leftHeat;
	int rightHeat;
	
	Default
	{
		Weapon.AmmoType "BDP_Cell";
		BDP_Weapon.MagCapacity 40;
		Weapon.SisterWeapon "BDP_PlasmaRifle";
		+WEAPON.POWERED_UP
		+BDP_Weapon.DUALWEAPON
		Tag "$BDPPLASMA";
		Inventory.AltHUDIcon "PRASA0";
		BDP_Weapon.handedness 2;
		BDP_Weapon.voiceacted 0;
		BDP_Weapon.DualWeapon "BDP_PlasmaRifle";
	}
	States
	{
	CacheSprites:
		4UPL A 0;
		3UPL A 0;
		2UPL A 0;
		4UPR A 0;
		3UPR A 0;
		2UPR A 0;
		Stop;
	TossItem: //30
		TNT1 A 6 A_DrawDualOverlays("DeselectLeft", "DeselectRight");
		TNT1 A 1;
		Wait;
	FinishToss:
		TNT1 A 6 A_DrawDualOverlays("SelectLeft", "SelectRight");
		Goto Ready;
	Kick: //18
		TNT1 A 4 A_DrawDualOverlays("KickLeft", "KickRight");
		TNT1 A 10;
		TNT1 A 4 A_DrawDualOverlays("KickLeftEnd", "KickRightEnd");
		Goto Ready;
	SlideKick: //25
		TNT1 A 4 A_DrawDualOverlays("KickLeft", "KickRight");
		TNT1 A 17;
		TNT1 A 4 A_DrawDualOverlays("KickLeftEnd", "KickRightEnd");
		Goto Ready;
	Sprint:
		TNT1 A 0 A_DrawDualOverlays("SprintLeft", "SprintRight");
		TNT1 A 4;
	SprintLoop:
		TNT1 A 1;
		Loop;
	SprintEnd:
		TNT1 A 0 A_DrawDualOverlays("SprintLeftEnd", "SprintRightEnd");
		TNT1 A 4;
		Stop;
	KickLeft:
		DUPL MNOP 1 A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		DUPL Q 1 A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		Wait;
	KickLeftEnd:
		DUPL PONM 1 A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		Goto ReadyLeft;
	KickRight:
		DUPR MNOP 1 A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR", true);
		DUPR Q 1 A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR", true);
		Wait;
	KickRightEnd:
		DUPR PONM 1 A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR", true);
		Goto ReadyRight;
	SprintLeft:
		DUPL MNOP 1
		{
			A_HandleDualFiring("FireLeft", sprinting:true);
			A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		}
	SprintLeftLoop:
		DUPL Q 1
		{
			A_HandleDualFiring("FireLeft", sprinting:true);
			A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		}
		Loop;
	SprintLeftEnd:
		DUPL PONM 1
		{
			A_HandleDualFiring("FireLeft", sprinting:true);
			A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		}
		Goto ReadyLeft;
	SprintRight:
		DUPR MNOP 1
		{
			A_HandleDualFiring("FireRight", sprinting:true);
			A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR", true);
		}
	SprintRightLoop:
		DUPR Q 1
		{
			A_HandleDualFiring("FireRight", sprinting:true);
			A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR", true);
		}
		Loop;
	SprintRightEnd:
		DUPR PONM 1
		{
			A_HandleDualFiring("FireRight", sprinting:true);
			A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR", true);
		}
		Goto ReadyRight;
	Ready:
		TNT1 A 1 A_WeaponReadyDX(WRF_ALLOWRELOAD|WRF_NOFIRE);
		Loop;
	ReadyLeft:
		TNT1 A 0 A_HandleDualFiring("FireLeft");
		DUPL A 1 A_LowSprites("3UPL", "4UPL", "2UPL", "DUPL");
		Loop;
	ReadyRight:
		TNT1 A 0 A_HandleDualFiring("FireRight");
		DUPR A 1 A_LowSprites("3UPR", "4UPR", "2UPR", "DUPR", true);
		Loop;
	SpecialAction:
		TNT1 A 0 A_EndDual();
		Goto Ready;
	Deselect:
		TNT1 A 7 A_DrawDualOverlays("DeselectLeft", "DeselectRight");
		TNT1 A 0 A_Lower();
		Wait;
	DeselectLeft:
		3UPL A 2 A_LowSprites("3UPL", "3UPL", "2UPL", "DUPL");
		DUPL IJKL 1;
		Stop;
	DeselectRight:
		DUPR IJKL 1;
		Stop;
	Select:
		TNT1 A 1
		{
			A_SetCrosshairDX("DPLSRet", 500);
			A_GunslingerMag();
		}
		DUPL A 0 A_StartSound("PLSDRAW");
		TNT1 A 0 A_DrawDualOverlays("SelectLeft", "SelectRight");
		TNT1 A 6;
		Goto Ready;
	SelectLeft:
		TNT1 A 2;
		DUPL LKJI 1;
		Goto ReadyLeft;
	SelectRight:
		DUPR LKJI 1;
		DUPR A 2 A_LowSprites("3UPR", "3UPR", "2UPR", "DUPR");
		Goto ReadyRight;
	SelectFromDualLeft:
		TNT1 A 4;
		TNT1 A 0 A_StartSound("PLSDRAW"); 
		DUPL VWXY 1;
		Goto ReadyLeft;
	SelectFromDualRight:
		DUPR RSTUVWXY 1;
		Goto ReadyRight;
	SelectFromDual:
		TNT1 A 0 {A_SetCrosshairDX("DPLSRet", 500); A_StartSound("foley/PlasmaInspectMisc", 10);}
		TNT1 A 0 A_DrawDualOverlays("SelectFromDualLeft", "SelectFromDualRight");
		TNT1 A 8;
		Goto Ready;
	MuzzleFlashLeft:
		DUPM A 1 Bright A_SetFrame(random(0, 1));
		Stop;
	MuzzleFlashRight:
		DUPM C 1 Bright A_SetFrame(random(2, 3));
		Stop;
	FireLeft:
		TNT1 A 0 A_CheckIfMag(label:"NoAmmoLeft");
		DUPL B 1 Bright
		{
			A_FireProjectileDX("Plasma_Ball", random(-4, 4), random(-4, 4), -8, 0, -8, false);
			A_TakeMag();
			A_FlashOverlay(-3, "MuzzleFlashLeft");
			A_FirePitchDown("PLSM9", 1, 30);
			A_AlertmonstersDX();
			A_QuadSound();
			A_GunLight(500, 2, 0, 128, 255);
			A_ZoomFactor(0.98);
			A_WeaponRecoil(0.7, frandom(-0.7, 0.7), 4);
			invoker.leftHeat++;
		}
		TNT1 A 0
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.blockFire = false;
		}
		DUPL D 1 A_ZoomFactor(1.0);
		DUPL E 1;
		TNT1 A 0 A_RefireDX("FireLeft");
		3UPL A 1;
		TNT1 A 0 A_JumpIf(invoker.leftHeat >= 5, "CooldownLeft");
		TNT1 A 0 {invoker.leftHeat = 0;}
		Goto ReadyLeft;
	FireRight:
		TNT1 A 0 A_CheckIfDualMag(label:"NoAmmoRight");
		DUPR B 1 Bright
		{
			A_FireProjectileDX("Plasma_Ball", random(-4, 4), random(-4, 4), 8, 0, -8, false);
			A_TakeDualMag();
			A_FlashOverlay(state:"MuzzleFlashRight");
			A_DualFirePitchDown("PLSM9", 1, 30);
			A_AlertmonstersDX();
			A_QuadSound();
			A_GunLight(500, 2, 0, 128, 255);
			A_ZoomFactor(0.98);
			A_WeaponRecoil(0.7, frandom(-0.7, 0.7), 4);
			invoker.rightHeat++;
		}
		TNT1 A 0
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.blockFire = false;
		}
		DUPR D 1 A_ZoomFactor(1.0);
		DUPR E 1;
		TNT1 A 0 A_RefireDX("FireRight");
		3UPR A 1;
		TNT1 A 0 A_JumpIf(invoker.rightHeat >= 5, "CooldownRight");
		TNT1 A 0 {invoker.rightHeat = 0;}
		Goto ReadyRight;
	CooldownLeft:
		TNT1 A 0
		{
			invoker.leftHeat = 0;
			A_StartSound("PLSCOOL", 3);
		}
		2RSL CDEFGH 1;
		2RSL IIII 4 A_FireProjectile("SmokeSpawner11",0,0,-28,16);
		2RSL HGFEDD 1;
		Goto ReadyLeft;
	CooldownRight:
		TNT1 A 0
		{
			invoker.rightHeat = 0;
			A_StartSound("PLSCOOL", 3);
		}
		2RSR CDEFGH 1;
		2RSR IIII 4 A_FireProjectile("SmokeSpawner11",0,0,28,16);
		2RSR HGFEDD 1;
		Goto ReadyRight;
	DoReload:
		TNT1 A 0
		{
			A_ClearOverlays(2,3);
			invoker.sideFired = false;
		}
		TNT1 A 6 A_DrawDualOverlays("DeselectLeft", "DeselectRight");
		TNT1 A 8;
		TNT1 A 0 A_JumpIf(DualMag() == 40, "ReloadLeft");
		TNT1 A 0
		{
			let spentmag = BDP_SpentMag(A_FireProjectileDX("BDP_SpentCell", random(75, 85), random(35, 55), 0, 25, -20));
			if(spentmag)
				spentmag.ammo = DualMag();
			A_DiscardDualMag();
			A_PlaySound("Plasma/Reload1", 5);
		}
		PRR1 ABCDE 1;
		PRSR GHH 1;
		TNT1 A 0 A_StartSound("foley/PlasmaInspectMisc", 10);
		PRSR I 20;
		PRSR JKLMNNO 1;
		TNT1 A 0 A_PlaySound("Plasma/Reload2", 5);
		PRSR OOOO 1;
		TNT1 A 0 A_ReloadDualMag();
		PRSR PQRSTTTTUV 1;
		PRR1 FGHBA 1;
		TNT1 A 8;
		TNT1 A 0 A_JumpIf(invoker.mag == 40 || CountInv("BDP_Cell") == 0, "FinishReload");
	ReloadLeft:
		TNT1 A 0
		{
			let spentmag = BDP_SpentMag(A_FireProjectileDX("BDP_SpentCell", random(-75, -85), random(35, 55), 0, 25, -20));
			if(spentmag)
				spentmag.ammo = invoker.mag;
			A_DiscardMag();
			A_PlaySound("Plasma/Reload1", 5);
		}
		PRR2 BCDEFGHH 1;
		TNT1 A 0 A_StartSound("foley/PlasmaInspectMisc", 10);
		PRR2 I 20;
		PRR2 JKLMNNO 1;
		TNT1 A 0 A_PlaySound("Plasma/Reload2", 5);
		PRR2 OOOO 1;
		TNT1 A 0 A_ReloadMag();
		PRR2 PQRSTTTTUVWXY 1;
		PRR2 CB 1;
		TNT1 A 8;
	FinishReload:
		TNT1 A 6 A_DrawDualOverlays("SelectLeft", "SelectRight");
		Goto Ready;
	NoAmmoLeft:
		TNT1 A 0 A_DryFire("BEPBEP");
		4UPL A 2;
		3UPL A 2;
		TNT1 A 0
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.blockFire = false;
		}
		4UPL A 2;
		3UPL A 2;
		TNT1 A 0 A_RefireDX("NoAmmoLeft");
		Goto ReadyLeft;
	NoAmmoRight:
		TNT1 A 0 A_DryFire("BEPBEP");
		4UPR A 2;
		3UPR A 2;
		TNT1 A 0
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.blockFire = false;
		}
		4UPR A 2;
		3UPR A 2;
		TNT1 A 0 A_RefireDX("NoAmmoRight");
		Goto ReadyRight;
	}
}
