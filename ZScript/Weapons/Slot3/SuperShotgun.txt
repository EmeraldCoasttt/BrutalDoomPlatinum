		Weapon.SlotNumber 3;
		Weapon.SlotPriority 3;
		Weapon.SelectionOrder 400;
		Weapon.AmmoGive 8;
		Weapon.AmmoType "BDP_Shell";
		BDP_Weapon.MagCapacity 2;
		BDP_Weapon.MagConversion (2,2);
		Inventory.PickupMessage "You got the Super Shotgun! (Slot 3)";
		Obituary "$OB_MPSSHOTGUN";
		Inventory.PickupSound "CLIPINSS";
		Tag "$BDPSSG";
		Inventory.AltHUDIcon "SGN2A0";
	}
	
	bool uraniumShells;
	
	States
	{
	CacheSprites:
		SHO4 A 0;
		Stop;
	TossItem: //8
		SHT5 EFGH 1;
		TNT1 A 1;
		Wait;
	FinishToss:
		SHT5 HGFE 1;
		Goto Ready;
	Kick: //18
		SH69 PQRS 1;
		SH69 T 10;
		SH69 SRQP 1;
		Goto Ready;
	SlideKick: //25
		SH69 PQRS 1;
		SH69 T 17;
		SH69 SRQP 1;
		Goto Ready;
	Taunt: //30
		SH69 PQRS 1;
		SH69 T 22;
		SH69 SRQP 1;
		Goto Ready;
	Ready:
		SHTZ A 1 A_WeaponReadyDX(WRF_ALLOWRELOAD,forcereload:true);
		Loop;
	Fidget:
		SHO3 ABCD 1 A_WeaponReadyDX(0,false);
		SHO3 E 3 A_WeaponReadyDX(0,false);
		SHO3 FGH 1 A_WeaponReadyDX(0,false);
		SHO3 I 1
		{
			if(invoker.uraniumshells)
				A_SetSprites("SHO4");
			return A_WeaponReadyDX(0,false);
		}
		TNT1 A 0 A_StartSound("weapons/sshoto");
		SHO3 JKL 1
		{
			if(invoker.uraniumshells)
				A_SetSprites("SHO4");
			return A_WeaponReadyDX(0,false);
		}
		SHO3 M 15
		{
			if(invoker.uraniumshells)
				A_SetSprites("SHO4");
			return A_WeaponReadyDX(0,false);
		}
	FidgetLoop:
		SHO3 M 1
		{
			if(invoker.uraniumshells)
				A_SetSprites("SHO4");
			return A_WeaponReadyDX(0,false);
		}
		TNT1 A 0 A_PressingReload("FidgetLoop");
		SHTZ A 0 A_StartSound("foley/SSGraiseAfterClose",10);
		SHO3 NOP 1
		{
			if(invoker.uraniumshells)
				A_SetSprites("SHO4");
			return A_WeaponReadyDX(0,false);

	SprintLoop:
		SH69 T 1 A_SetSprites("TNT1A0",1);

	SprintEnd:

		TNT1 A 1;
		TNT1 A 0 ChangeAmmoIcon1("SHELA0");

		Wait;
	Select:

		{
			A_SetCrosshairDX("SSGRet", 400);
			A_SetTargeterCrosshair("MEATRet",1600,1.0);
			if(invoker.uraniumShells)
				ChangeAmmoIcon1("SHEL]0");
			A_GunSlingerMag();
		}

		Goto Ready;

		SHOM AB 1 Bright;
		Stop;
	Fire:
		TNT1 A 0 A_CheckIfMag(2,force:true);
		SHOF A 1 Bright
		{
			A_CutMeathook();
			if(invoker.uraniumShells)
			{
				for(int i = 0; i < 14; i++)
				{
					A_FirePiercingBullet(3, "UraniumPuff", 10, 3, 0, -4);
				}

		SSGS CDEF 1;
		Stop;

		}

			A_StartSound("weapons/sshotl");
			A_ReloadMag();
			if(invoker.uraniumShells)
			{
				invoker.uraniumShells = false;
				ChangeAmmoIcon1("SHELA0");
				invoker.magConversion = (2, 2);
			}

			if(invoker.mag < 2)

		}

	Spawn:
		TNT1 A 0 NoDelay A_Jump(256, "Spawn1", "Spawn2", "Spawn3");
		Loop;
	Spawn1:
		SGN2 B 1 A_JumpIfInventory("AmmoTaken", 1, "SpawnNoAmmo");
		Loop;
	Spawn2:
		SGN2 C 1 A_JumpIfInventory("AmmoTaken", 1, "SpawnNoAmmo");

	Spawn3:
		SGN2 D 1 A_JumpIfInventory("AmmoTaken", 1, "SpawnNoAmmo");
		Loop;
	Spawn4:
		SGN2 E 1 A_JumpIfInventory("AmmoTaken", 1, "SpawnNoAmmo");
		Loop;
	Spawn5:
		SGN2 F 1 A_JumpIfInventory("AmmoTaken", 1, "SpawnNoAmmo");
		Loop;
	Spawn6:
		SGN2 G 1 A_JumpIfInventory("AmmoTaken", 1, "SpawnNoAmmo");
		Loop;
	SpawnNoAmmo:
		SGN2 A -1;
		Stop;
	AltFire:
		TNT1 A 0 A_trymeathook(1600);
		Goto Ready;
	SpecialAction:
		TNT1 A 0
		{
			if(invoker.uraniumShells)
				A_Print("\cgBuckshot", 1);

				A_Print("\cdUranium Shells", 1);

				A_SetFrame(16);
		}

			A_TakeMag(2);
			A_StartSound("foley/SSGshellEject",10,0,0.5);
			A_FireProjectileDX(invoker.uraniumShells ? "BDP_UnspentUraniumShell" : "BDP_UnspentSSGShell", random(80, 100), random(-50, -70), -4, 25, -14);
			A_FireProjectileDX(invoker.uraniumShells ? "BDP_UnspentUraniumShell" : "BDP_UnspentSSGShell", random(80, 100), random(-50, -70), -2, 25, -13);
		}
		TNT1 A 0 A_JumpIf(!invoker.uraniumShells, "ContinueReloadUraniumShells");
		Goto ContinueReload;
	}

class DragonsBreathSparkle : Actor
{
	Default
	{
		Radius 2;
		Height 2;
		Speed 60;
		DamageFunction(1);
		Projectile;
		+RANDOMIZE
		RenderStyle "Add";
		Alpha 0.75;
		Obituary "$OB_MPPLASMARIFLE";
		Gravity 0.3;
		damagetype "Fire";
		+forcexybillboard;
		+ripper;
	}
	
	Override void beginplay()
	{
		Speed = frandom(90,120);
		float potentialscale = frandom(0.05,0.09);
		A_setscale(potentialscale,potentialscale);
		Super.beginplay();
	}
	Override void Tick()
	{
		if(GetAge() > 1 && GetAge() < 20)
		{
			FSpawnParticleParams dbtrail;
			dbtrail.texture = TexMan.CheckForTexture("SPRKA0");
			dbtrail.color1 = "";
			dbtrail.style = STYLE_Add;
			dbtrail.flags = SPF_FULLBRIGHT;
			dbtrail.startalpha = alpha;
			dbtrail.size = frandom(8,12);
			dbtrail.sizestep = -0.5;
			dbtrail.lifetime = 24; 
			dbtrail.pos = pos;
			Level.SpawnParticle(dbtrail);
		}
		Super.Tick();
	}
	States
	{
	Spawn:
		TNT1 A 0;
		TNT1 A 0 A_jump(200,"Spawn2");
		TNT1 B 0;
		Spawn2:
		TNT1 "#" 1;
		SPRK "#" Random(0,4) Bright;
		TNT1 "#" 0
		{
			vel *= 0.1;
			bnogravity = false;
			bmissile = false;
			gravity = frandom(0.05,0.3);
		}
		SPRK "#" Random(20,120) Bright;
	Death:
		SPRK "#" 1 bright A_FadeOut (0.05);
		loop;
	}
	
	Override int SpecialMissileHit(Actor victim) 
		{
			If(victim && target && victim != target && !(victim is "BDP_Headshot") && victim.bshootable)
			{
				If(!Victim.findinventory("firetic"))
				{
					Victim.giveinventory("firetic",1);
				}
				Actor Flameactor = Victim.Findinventory("Firetic");
				If(Flameactor)
				{
					let flamer = Firetic(flameactor);
					if(flamer)
					{
						flamer.flamecauser = target;
						if(flamer.duration < 90)
						flamer.duration = 90;
						flamer.duration += 0.05;
					}
				
				}
				
			}
		
			Return -1;
		}
		
}

class UraniumItem : Inventory
{
	int count;
	
	override void DoEffect()
	{
		Super.DoEffect();
		while(owner && count >= 7)
		{
			owner.A_startsound("FAREXPL", 0);
			owner.A_SpawnItemEx("WhiteShockwaveSmall", zofs:32);
			owner.A_CustomMissile ("ExplosionSmoke", 32, 0, random (0, 360), 2, random (0, 360));
			owner.A_SpawnItemEx ("UnderwaterExplosion", zofs:32);
			for(int i = 0; i < 4; i++)
			{
				owner.A_SpawnItemEx("UraniumExplosion", random(-32, 32), random(-32, 32), random(16, 48));
				owner.A_SpawnItemEx("VehicleSmoke", random(-32, 32), random(-32, 32), random(16, 48));
			}
			owner.TriggerPainChance("Normal", true);
			owner.RadiusAttack(self, 100, 64);
			count -= 7;
		}
		if(owner)
			owner.SetInventory("UraniumItem", 0);
	}
}
