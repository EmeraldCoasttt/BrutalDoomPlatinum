class BDP_WeaponDropper : Actor
{
	class<BDP_DroppedWeapon> weaponToDrop;
	property WeaponToDrop : weaponToDrop;
	
	Default
	{
		Radius 0;
		Height 0;
		+THRUACTORS
	}
	States
	{
		Spawn:
		TNT1 A 0 NoDelay A_CustomMissile(WeaponToDrop, 20, 0, random (0, 360), 2, random (30, 50));
		Stop;
	}
}

class BDP_DroppedWeapon : Actor
{
	Default
	{
		speed 6;
		health 1;
		radius 16;
		height 4;
		Gravity 0.9;
		damage 0;
		XScale 0.7;
		YScale 0.8;
		+ROLLSPRITE
		+ROLLCENTER
		+THRUACTORS
		+MISSILE
		+NOTELEPORT
		+NOBLOCKMAP
		+BLOODLESSIMPACT 
		+FORCEXYBILLBOARD
		+NODAMAGETHRUST
		+MOVEWITHSECTOR
		Mass 1;
		BounceType "Doom";
		BounceFactor 0.4;
		WallBounceFactor 0.2;
	}
	action void A_SpawnDroppedWeapon(name weapontype, int ammonum)
	{
		Weapon weap2drop = Weapon(Spawn(weapontype,pos));
		if (weap2drop)
		{
			weap2drop.ammoGive1 = ammonum;
		}
	}
}
class BDP_PistolDrop : BDP_WeaponDropper {Default {BDP_WeaponDropper.WeaponToDrop "BDP_DroppedPistol";}}
class BDP_RifleDrop : BDP_WeaponDropper {Default {BDP_WeaponDropper.WeaponToDrop "BDP_DroppedRifle";}}
class BDP_ShotgunDrop : BDP_WeaponDropper { Default {BDP_WeaponDropper.WeaponToDrop "BDP_DroppedShotgun";}}
class BDP_SMGDrop : BDP_WeaponDropper {Default {BDP_WeaponDropper.WeaponToDrop "BDP_DroppedSubmachineGun";}}
class BDP_MinigunDrop : BDP_WeaponDropper {Default {BDP_WeaponDropper.WeaponToDrop "BDP_DroppedMinigun";}}

class BDP_DroppedRifle : BDP_DroppedWeapon
{
	States
	{
	Spawn:
		RIFL A 1 {A_SetRoll(roll + 11.25);bThruActors = 0;}
		Loop;
	Death:
		TNT1 A 0 A_SpawnDroppedWeapon("BDP_AssaultRifle",15);

	}
