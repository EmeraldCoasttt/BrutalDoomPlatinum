class BDP_SpentCasing : Actor
{
	Default
	{
		Height 2;
		Radius 2;
		Speed 6;
		Scale 0.15;
		BounceType "Doom";
		+WINDTHRUST
		+CLIENTSIDEONLY
		+MOVEWITHSECTOR
		+MISSILE
		+NOBLOCKMAP
		+NOTELEPORT
		+FORCEXYBILLBOARD
		+ROLLSPRITE
		+ROLLCENTER
		Mass 0;
		BounceFactor 0.4;
		BounceSound "weapons/casing";
	}
	States
	{
	Spawn:
		TNT1 A 0;
	Fly:
		#### A 1 A_SetRoll(roll+22.5);
		Loop;
	Death:
		#### A 0
		{
			bXFLIP = random(0,1);
			A_SetRoll(0);
		}
		#### A 0 A_Jump(256, "Rest1", "Rest2", "Rest3");
		Goto Rest1;
	Rest1:
		#### B 400;
		Goto Fadeout;
	Rest2:
		#### C 400;
		Goto Fadeout;
	Rest3:
		#### D 400;
		Goto Fadeout;
	FadeOut:
		#### # 1 A_FadeOut(0.02);
		Loop;
	}
}

class BDP_SpentMag : Actor abstract
{
	Default
	{
		+WINDTHRUST
		+CLIENTSIDEONLY
		+MOVEWITHSECTOR
		+MISSILE
		+NOBLOCKMAP
		+NOTELEPORT
		+FORCEXYBILLBOARD
		+ROLLSPRITE
		+ROLLCENTER
		Height 2;
		Radius 2;
		Speed 4;
		Mass 1;
		BounceType "Doom";
		DeathSound "weapons/largemagdrop";
		BounceSound "weapons/largemagdrop";
	}
	
	int ammo;
	
	override void PostBeginPlay()
	{
		Super.PostBeginplay();
		SetSprites();
	}
	
	virtual void SetSprites()
	{
		console.printf("you shouldn't be seeing this");
	}
	
	States
	{
	Spawn:
		TNT1 A 0;
	Fly:
		#### # 1 A_SetRoll(roll+11.25);
		Loop;
	Death:
		#### # 350
		{
			if(random(0, 1) == 1)
			{
				A_ChangeFlag("XFLIP",1);
			}
			A_SetRoll(0);
		}
	FadeOut:
		#### # 1 A_FadeOut(0.02);
		Loop;
	}
}
