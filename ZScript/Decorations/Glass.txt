class BDP_BreakableWindow : Actor
{
	Default
	{
		+NOBLOCKMAP
		-SOLID
		XScale 0.74;
		YScale 3.0;
	}
	States
	{
	Spawn:
		TNT1 A 0 NoDelay
		{
			A_SpawnItemEx("BDP_WindowGlass", flags:SXF_SETMASTER);
			A_SpawnItemEx("BDP_WindowGlass", angle:180, SXF_SETMASTER);
			A_SpawnItemEx("BDP_UnbreakableWindowCollider", yofs:-30);
			A_SpawnItemEx("BDP_UnbreakableWindowCollider", yofs:30);
			for(int i = -3; i < 4; i++)
				A_SpawnItemEx("BDP_WindowCollider", yofs:i * 8, flags:SXF_SETMASTER);
		}
		XXX1 A -1;
		Stop;
	}
}

class BDP_WindowCollider : Actor
{
	Default
	{
		+SOLID
		+SHOOTABLE
		+DONTTHRUST
		+NOBLOOD
		Health 1;
		Radius 4;
		Height 216;
		DamageFactor "Avoid", 0.0;
	}
	States
	{
	Spawn:
		TNT1 A -1;
		Stop;
	Death:
		TNT1 A 0
		{
			master.A_KillChildren();
		}
		Stop;
	}
}

class BDP_WindowGlass : BDP_WindowCollider
{
	Default
	{
		XScale 0.74;
		YScale 3.0;
		Renderstyle "Translucent";
		Alpha 0.65;
	}
	States
	{
	Spawn:
		BRGL A -1;
		Stop;
	Death:
		TNT1 A 0
		{
			master.A_KillChildren();
			A_NoBlocking();
			A_StartSound("WINBREA");
			for(int i = 0; i < 16; i++)
			{
				string actor = "LargeGlassParticle"..random(1, 4);
				A_SpawnProjectile(actor, random(5, 200), random(-25, 25), random(-45, 45), CMF_AIMDIRECTION, random(-20, -70));
			}
		}
		XXX1 B -1;
		Stop;
	}
}

class BDP_UnbreakableWindowCollider : BDP_WindowCollider
{
	Default
	{
		+NODAMAGE
		Radius 2;
	}
}
