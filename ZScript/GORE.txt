enum EGoreGroups
{
	GROUP_All,
	GROUP_Face,
	GROUP_Head,
	GROUP_LeftArm,
	GROUP_RightArm,
	GROUP_LeftLeg,
	GROUP_RightLeg
}

extend class BDP_Actor
{
	array<EGoreGroups> bodyParts;
	name sufferDeath;

	override void PostBeginPlay()
	{
		super.PostBeginPlay();
		bodyParts.Push(GROUP_Face);
		bodyParts.Push(GROUP_Head);
		bodyParts.Push(GROUP_LeftArm);
		bodyParts.Push(GROUP_RightArm);
		bodyParts.Push(GROUP_LeftLeg);
		bodyParts.Push(GROUP_RightLeg);
	}

	void A_SpawnGore(class<BDP_GoreGroup> groupClass, EGoreGroups group, double velocity, int minAngle = 0, int maxAngle = 360, double zpos = 0.0)
	{
		BDP_GoreGroup g = BDP_GoreGroup.Create(groupClass);
		array<class<Actor>> actors;
		if(group == GROUP_All)
		{
			for(int i = 0; i < bodyParts.Size(); i++)
				g.AddGoreGroups(bodyParts[i]);
			bodyParts.Clear();
		}
		else
		{
			int findPart = bodyParts.Find(group);
			if(findPart == bodyParts.Size()) return;
			g.AddGoreGroups(group);
			bodyParts.Delete(findPart);
		}
		g.GetGoreGroups(actors);
		for(int i = 0; i < actors.size(); i++)
		{
			let gib = Spawn(actors[i], (pos.xy, pos.z + zpos));
			gib.Vel3DFromAngle(velocity * frandom(0.5,1.5), angle + random(minAngle, maxAngle), random(-15, -75));
		}
	}
	
	override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle)
	{
		if(sufferDeath != 'None' && health - ApplyDamageFactor(mod, damage) < 1)
		{
			mod = sufferDeath;
			sufferdeath == '';
			bNOPAIN = default.bNOPAIN;
			bNOTARGET = default.bNOTARGET;
		}
		super.DamageMobj(inflictor, source, damage, mod, flags, angle);
		return damage;
	}
	
	void A_SufferState(name nextState, int bodyHealth = 50)
	{
		sufferDeath = nextState;
		health = bodyHealth;
		height = default.height;
		bSHOOTABLE = true;
		bNOPAIN = true;
	}
	
	void A_ResetSuffer()
	{
		sufferDeath = '';
		bSHOOTABLE = false;
		health = 0;
		bNOPAIN = default.bNOPAIN;
		bNOTARGET = default.bNOTARGET;
	}
	
	void A_CorpseState(int bodyHealth = 50)
	{
		A_SufferState('CorpseGibbed', bodyHealth);
	}
}

class BDP_GoreGroup
{
	array<class<Actor>> actors;
	
	static BDP_GoreGroup Create(class<BDP_GoreGroup> group)
	{
		BDP_GoreGroup g = BDP_GoreGroup(New(group));
		return g;
	}
	
	virtual void AddGoreGroups(EGoreGroups group)
	{
	}
	
	virtual void GetGoreGroups(in out array<class<Actor>> actors)
	{
	}
	
	void AddGoreActor(class<Actor> act, int amt = 1)
	{
		for(int i = 0; i < amt; i++)
		{
			actors.push(act);
		}
	}
}

class BDP_ZombieguyGoreGroups : BDP_GoreGroup
{
	override void AddGoreGroups(EGoreGroups group)
	{
		switch(group)
		{
		case GROUP_Face:
		case GROUP_Head:
			//AddGoreActor("PlasmaBall", 400);
			AddGoreActor("BDP_BrainGib", 2);
			AddGoreActor("BDP_EyeGib");
			AddGoreActor("BDP_ToothGib", 6);
			AddGoreActor("Blood", 3);
			AddGoreActor("NashGoreBloodSpurt", 6);
			break;
		case GROUP_LeftArm:
			break;
		case GROUP_RightArm:
			break;
		case GROUP_LeftLeg:
			break;
		case GROUP_RightLeg:
			break;
		}
	}
	
	override void GetGoreGroups(in out array<class<Actor>> actors)
	{
		actors = self.actors;
	}
}

	}

	}

	Class BDP_CorpseInventory : inventory
	{

		Default
		{
		+INVENTORY.UNDROPPABLE
		+INVENTORY.UNTOSSABLE
		inventory.maxamount 1;
		}
		bool bsquished;
		
		Override void doeffect()
		{
			int spawnFlags = SXF_USEBLOODCOLOR | SXF_NOCHECKPOSITION;

			double crushHeight = (Owner.Pos.Z + Owner.Height) + Owner.GetDeathHeight();

			if (Owner.CeilingZ <= crushHeight && Owner.Pos.Z <= Owner.FloorZ && !bSquished)
			{
				owner.setstatelabel("Death.CrushedBodies");

				bSquished = true;
			}
			Super.doeffect();
				
		}
	}
	
	Class BDP_Corpse : Actor
	{
	
	
		Override void tick()
		{
			
			If(health >= 1 && !findinventory("BDP_CorpseInventory") && bshootable)
			{
				Giveinventory("BDP_CorpseInventory",1);
			}
			Super.tick();
			
		}
	}
