extend class BDP_Actor
{
	action void A_SpawnGore(class<BDP_GoreGroup> group, double velocity, double xposmult = 1.0, double yposmult = 1.0, double zposmult = 1.0)
	{
		BDP_GoreGroup g = BDP_GoreGroup.Create(group);
		for(int i = 0; i < g.actors.size(); i++)
		{
			let [result,actor] = A_SpawnItemEx(g.actors[i],g.xpos[i] * xposmult,g.ypos[i] * yposmult,g.zpos[i] * zposmult);
			if(g.flipsprite[i] > -1) actor.bSPRITEFLIP = g.flipsprite[i];
			actor.Vel3DFromAngle(velocity * frandom(0.8,1.2), angle + g.angle[i], g.pitch[i]);
		}
	}
}

class BDP_GoreGroup
{
	array<class<Actor> > actors;
	array<double> xpos;
	array<double> ypos;
	array<double> zpos;
	array<double> angle;
	array<double> pitch;
	array<bool> flipsprite;
	
	static BDP_GoreGroup Create(class<BDP_GoreGroup> group)
	{
		BDP_GoreGroup g = BDP_GoreGroup(New(group));
		if(g)
		{
			g.SetGoreGroups();
		}
		return g;
	}
	
	virtual void SetGoreGroups()
	{
	}
	
	void AddGoreActor(class<Actor> act, double x, double y, double z, double dirxy, double dirz, int flip = -1)
	{
		actors.push(act);
		xpos.push(x);
		ypos.push(y);
		zpos.push(z);
		angle.push(dirxy);
		pitch.push(dirz);
		flipsprite.push(flip);
	}
}

class BDP_BaseGib : NashGoreGibs
{
	bool CanStretch;
	bool BloodSplashes;
	property CanStretch : CanStretch;
	property BloodSplashes : BloodSplashes;
	
	Default
	{
		BDP_BaseGib.CanStretch false;
		BDP_BaseGib.BloodSplashes true;
	}
	States
	{
	Bounce.Floor:
		"####" "#" 0
		{
			bouncefactor *= 0.73;

			// boing boing
			if (WaterLevel == 0 && CanStretch)
				Scale.X = Default.Scale.X * 2.224;

			// spawn one large blood splash upon bounce impact
			if(BloodSplashes)
			{
				Actor bloodSplash;
				bool result;
				[result, bloodSplash] = A_SpawnItemEx("NashGoreBloodFloorSplash", flags: NashGoreBloodBase.BLOOD_FLAGS);
				if (bloodSplash)
					bloodSplash.Scale = (bloodSplash.Scale.X * 2.0, bloodSplash.Scale.Y * 2.0);
			}
		}
		Goto SpawnLoop;
	Death:
		"####" B 0
		{
			A_SetRoll(0);

			// clip gib into terrain
			let t = GetFloorTerrain();

			// clip by 2 map units; ignore the terrain's FootClip value
			if (t && t.FootClip > 0)
				FloorClip = 2.0;

			if(bloodSplashes)
			{
				class<Actor> cls;
				if (nashgore_bloodtype == 0)
					cls = "NashGoreBloodSpot";
				else if (nashgore_bloodtype == 1)
					cls = "NashGoreBloodSpotClassic";
				int failChance = 150;
				A_SpawnItemEx(cls, flags: (NashGoreBloodBase.BLOOD_FLAGS | SXF_TRANSFERPOINTERS) & ~SXF_NOCHECKPOSITION, failChance);
			}

			if (NashGoreHandler.CheckForVoxels())
				NashGoreGameplayStatics.AlignToPlane(self);

			NashGoreGameplayStatics.FixZFighting(self);
		}
		"####" BBBBBBBBBB 1
		{
			if (WaterLevel < 3 && BloodSplashes)
			{
				int failChance = 175;
				A_SpawnItemEx("NashGoreBloodFloorSplashSpawner",
					0, 0, 0,
					frandom[rnd_GibFloorSplash](-4.0, 4.0), frandom[rnd_GibFloorSplash](-4.0, 4.0), frandom[rnd_GibFloorSplash](1.0, 4.0),
					frandom[rnd_GibFloorSplash](0.0, 360.0), NashGoreBloodBase.BLOOD_FLAGS, failChance);
			}
		}
		"####" B -1;
		Stop;
	}
}

class BDP_ShootableGib : BDP_BaseGib
{
	Default

		Height 10;

	}

		"####" A 0 A_JumpIf(health <= 0,"RealDeath");
		"####" A 0 {
			bSHOOTABLE = true;
			bMISSILE = false;
			A_ChangeLinkFlags(false,false);
		}
		Goto Super::Death;
	RealDeath:
		TNT1 A 0 A_StartSound("misc/xdeath5");
		Stop;
	}
	
	action void A_SpawnSubGibs(string toSpawn, double x, double y, double z)
	{
		let [a, b] = A_SpawnItemEx(toSpawn,xvel:x * frandom(0.9,1.1),yvel:(bSPRITEFLIP ? -y : y) * frandom(0.9,1.1),zvel:z * frandom(0.9,1.1));
		b.bSPRITEFLIP = bSPRITEFLIP;
	}
}

	Class BDP_CorpseInventory : inventory
	{

		Default
		{
		+INVENTORY.UNDROPPABLE
		+INVENTORY.UNTOSSABLE
		inventory.maxamount 1;
		}
		bool bsquished;
		
		Override void doeffect()
		{
			int spawnFlags = SXF_USEBLOODCOLOR | SXF_NOCHECKPOSITION;

			double crushHeight = (Owner.Pos.Z + Owner.Height) + Owner.GetDeathHeight();

			if (Owner.CeilingZ <= crushHeight && Owner.Pos.Z <= Owner.FloorZ && !bSquished)
			{
				owner.setstatelabel("Death.CrushedBodies");

				bSquished = true;
			}
			Super.doeffect();
				
		}
	}
	
	Class BDP_Corpse : Actor
	{
	
	
		Override void tick()
		{
			
			If(health >= 1 && !findinventory("BDP_CorpseInventory") && bshootable)
			{
				Giveinventory("BDP_CorpseInventory",1);
			}
			Super.tick();
			
		}
	}
