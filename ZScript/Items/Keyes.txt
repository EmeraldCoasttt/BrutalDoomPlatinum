class BDP_Key : Inventory abstract
{
	Default
	{
		Radius 20;
		Height 16;
		Inventory.MaxAmount 0;
		Inventory.PickupSound "items/keycard";
		+NOTDMATCH
		+INVENTORY.AUTOACTIVATE
	}
	
	class<Key> keyActor;
	property KeyActor : keyActor;
	
	int keyID; 
	property KeyID : keyID;
	
	override bool Use(bool pickup)
	{
		owner.GiveInventory(keyActor, 1);
		return true;
	}
	
	override void BeginPlay()
	{
		if(random(0, 20) == 20)
			frame = 2;
		/*else
			frame = randompick(0, 0, 5, 7);*/
		bXFLIP = random(0, 1);
		ACS_NamedExecuteAlways("areascantid" ,0, keyID);
		super.BeginPlay();
	}
}

class BDP_SkullKey : BDP_Key abstract
{
	Default
	{
		Inventory.PickupSound "items/skullkey";
	}
	
	override void BeginPlay()
	{
		super.BeginPlay();
		if(random(0, 20) == 20)
			frame = randompick(8, 10);
		else
			frame = randompick(0, 0, 0, 2, 4, 6);
	}
}

class BDP_BlueCard : BDP_Key replaces BlueCard
{
	Default
	{
		Tag "Blue Keycard";
		Inventory.Pickupmessage "$GOTBLUECARD";
		Inventory.Icon "STKEYS0";
		BDP_Key.KeyID 0;
		BDP_Key.KeyActor "BlueCard";
	}
	States
	{
	Spawn:
		BKEY # 10 {frame--;}
		BKEY # 10 BRIGHT {frame++;}
		Loop;
	}
}
