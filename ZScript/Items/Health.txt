class BDP_HealthBonus : Health Replaces HealthBonus
{
	Default
	{
		+COUNTITEM
		+INVENTORY.ALWAYSPICKUP
		+INVENTORY.NEVERRESPAWN
		Inventory.Amount 2;
		Inventory.MaxAmount 300;
		Tag "Health Bonus";
		Inventory.PickupMessage "$GOTHTHBONUS";
		Health.LowMessage 10, "Picked up a health bonus. It's better than nothing...";
		Inventory.PickupSound "items/healthbonus";
		BDP_HealthBonus.ScoreAmount 10;
	}
	
	mixin BDP_ScoreMixin;
	
	States
	{
	Spawn:
		BON1 A 0 NoDelay
		{
			bXFLIP = random(0, 1);
			return A_Jump(256, "Spawn1", "Spawn2");
		}
	Spawn1:
		BON1 AABBCCDDCCBB 2 A_SpawnItemEx("BlueFlareSmallItem", zofs:7);
		Loop;
	Spawn2:
		BON1 EEFFGGHHGGFF 2 A_SpawnItemEx("BlueFlareSmallItem", zofs:7);
		Loop;
	}
}

class BDP_LightArmor : BasicArmorPickup Replaces GreenArmor

	{

		Inventory.Pickupmessage "Picked up the security armor.";

		Armor.SaveAmount 100;

class BDP_Stimpack : Health replaces Stimpack
{
	Default
	{
		Inventory.Amount 10;
		Inventory.PickupMessage "$GOTSTIM";
		Health.LowMessage 10, "At least it'll help ease the pain...";
		Inventory.PickupSound "GOTHPS";
		BDP_Stimpack.ScoreAmount 50;
	}
	
	mixin BDP_ScoreMixin;
	
	override void BeginPlay()
	{
		frame = random(0, 2);
		if(BDP_Handler.getMedikitColor()) frame += 3;
		super.BeginPlay();
	}
	
	States
	{
	Spawn:
		STIM # -1;
		Stop;
	}
}
