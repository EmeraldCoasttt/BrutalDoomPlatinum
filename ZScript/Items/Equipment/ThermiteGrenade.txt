class BDP_ThermiteGrenade : BDP_Equipment
{
	Default
	{
		Tag "Thermite Grenade";
		Inventory.Icon "BS2XA0";
		Inventory.PickupMessage "Picked up a thermite grenade.";
		Inventory.PickupSound "GRNPIN";
		BDP_Equipment.Cooldown 30;
		Scale 0.75;
	}
	
	override bool UseEffect()
	{
		owner.TakeInventory("BDP_ThermiteGrenadeStates", 1);
		owner.GiveInventory("BDP_ThermiteGrenadeStates", 1);
		return true;
	}
	
	States
	{
	Spawn:
		BS2X A -1;
		Stop;
	}
}

class BDP_ThermiteGrenadeStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	override void Tick()
	{
		return;
	}
	
	action void A_QuadSound()
	{	
		If(invoker.owner.findinventory("BDP_PowerQuadDamage"))
		{
			invoker.owner.a_startsound("items/quaddamage/active", 103);
		}
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.canLedgeClimb = false;
			A_ClearOverlays(49,50);
			A_Overlay(90, "TossGrenade");
			player.SetPSprite(1, player.readyweapon.ResolveState("TossItem"));
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	TossGrenade:
		TNT1 A 5;
		GRTH ABCDEF 1;
		TNT1 A 0 A_StartSound ("GRNPIN");
		GRTH FGHIJK 1;
		TNT1 A 2;
		TNT1 A 0 A_StartSound ("CS16uwu");
		TNT1 A 0 A_Jump(2,2);
		TNT1 A 0 A_StartSound ("GRNTOSS");
		GRTH LM 1;
		TNT1 A 0
		{
			A_FireProjectile("FireGrenade");
			A_QuadSound();
		}
		GRTH NOP 1;
		TNT1 A 5;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			TakeInventory("BDP_ThermiteGrenadeStates", 1);
		}
		Stop;
	}
}

class FireGrenade : Actor
{
	Default
	{
		Radius 4;
		Height 4;
		Speed 40;
		Gravity 0.3;
		DamageFunction 0;
		Projectile;
		Scale 0.75;
		Obituary "%o was stuck by a thermite grenade.";
		BounceType "Doom";
		BounceCount 5;
		BounceSound "Weapons/GrenadeBounce";
		DeathSound "Weapons/GrenadeBounce";
		-NOGRAVITY
		-ALLOWBOUNCEONACTORS
		+HITTRACER
		+FORCEXYBILLBOARD
		+CANBOUNCEWATER
	}
	

		
	
	Vector3 stickOfs;
	double stickAngle;
	
	int sawtimer;
	int sawtimer2;
	bool hasexploded;
	bool stuck;
	bool ihitsomething;
	
	Override void tick()
	{
		Super.tick();
		A_spawnprojectile("SmallFlameTrails", 0, 0, random (0, 360), 2, random (-70, -110));
		A_spawnprojectile("firegrenade_debris", 0, 0, random (0, 360), 2, random (0, 360));
		A_spawnprojectile("firegrenade_debris", 0, 0, random (0, 360), 2, random (0, 360));
		A_spawnprojectile("firegrenade_debris", 0, 0, random (0, 360), 2, random (0, 360));
		sawtimer++;
		sawtimer2++;
		If(sawtimer >= 10)
		{
			sawtimer = 0;
			If(tracer && target)
			{
				
				tracer.damagemobj(Self,Target,15,"Fire");
				
			}
			A_explode(15,200,XF_HURTSOURCE | XF_THRUSTLESS, false, DAMAGETYPE: "Fire");
		}
		If(sawtimer2 >= 245)
		{
			A_stopsound(4);
			Destroy();
		}
	
	}
	
	
	Override void beginplay()
		{
		//tracer = null;
		super.beginplay();
		//savedangle = angle;
		}
	
	
	States
	{
	Spawn:
		BS2X ABCD 1
		{
		A_startsound("THRMT",4,CHANF_LOOPING);
		//savedheight = floorz;
		tracer = null;
		}
		Loop;
		
		
		

		Xdeath:
		TNT1 A 0;
		//TNT1 A 0 A_startsound("StickyGrenade/Hit");
		TNT1 A 0 
		{
		ihitsomething = true;
		stuck = true;
		//tracer = victim;
			If(tracer)
			{
				tracer.a_pain();
				stickOfs = tracer.Vec3To(self);
					stickAngle = tracer.angle;
		busebouncestate = false;
		}
		If(!tracer)
		{
		setstatelabel("Death");
		}
		}
		Stuck:
		BS2X A 1 
		{
		
		If(!tracer || tracer.health < 1)
		{
		setstatelabel("Death");
		}


			If(tracer && tracer.health > 0)
			{
			/*
			If(tracer.health < 200 && !tracer.findinventory("melee_attacks"))
			{
			//tracer.angle = (tracer.angle + frandom(-40,40));
		//	tracer.a_stop();
		//	tracer.a_recoil(-7);
			tracer.reactiontime = 10;
			}
			*/
			
				double angDiff = DeltaAngle(stickAngle, tracer.angle);
					if (angDiff)
					{
						stickOfs.xy = RotateVector(stickOfs.xy, angDiff);
						angle += angDiff;
					}

				SetOrigin(tracer.Vec3Offset(stickOfs.x, stickOfs.y, stickOfs.z), true);
				stickAngle = tracer.angle;
				//tracer.a_stop();
			
				//Warp(tracer,calculatingpos.x,calculatingpos.y,calculatedpos.z,savedangle,WARPF_NOCHECKPOSITION | WARPF_ABSOLUTEANGLE | WARPF_INTERPOLATE | WARPF_TOFLOOR);
				//angle = savedangle;
			}
		
			
		}
		LOOP;
		
		death:
		TNT1 A 0 
		{
		ihitsomething = true;
		//A_startsound("StickyGrenade/Hit");
		tracer = null;
		}
		
		TNT1 A 0
		{
		busebouncestate = false;
		bnogravity = true;
		}
		StayDead:
		BS2X A 1;
		LOOP;
		

		
		
	
	ExplodeTime:
	TNT1 A 0;
		TNT1 A 10;
		stop;
		
	Fall:
		 //AXEG A 0 A_SpawnItemEX("Sparks");
		//AXEG A 0 A_startSound("AXECLN", 6);
		TNT1 A 0
		{
		tracer = null;
		bnogravity = false;
		}
	Goto Death;
	
	
	Bounce:
	TNT1 A 0
	{
	tracer = null;
	bhitowner = TRUE;
	}
	Goto spawn;
	
	}
}
