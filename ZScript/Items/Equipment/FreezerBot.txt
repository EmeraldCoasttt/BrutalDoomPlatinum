class BDP_FreezerBot : BDP_Equipment
{
	Default
	{
		Tag "Freezer Bot";
		Inventory.Icon "FZB1A1";
		Inventory.PickupMessage "Picked up a freezer bot.";
		Inventory.PickupSound "freezebot/talk";
		BDP_Equipment.Cooldown 120;
		Scale 0.8;
	}
	
	FreezeBot buddy;
	int deployState;
	
	override void DoEffect()
	{
		if(deployState == 2 && (!buddy || buddy.health < 1))
		{
			deployState = 0;
			TriggerCooldown();
		}
		super.DoEffect();
	}
	
	override bool UseEffect()
	{
		if(deployState == 0)
		{
			owner.TakeInventory("BDP_FreezerBotStates", 1);
			owner.GiveInventory("BDP_FreezerBotStates", 1);
			deployState = 1;
		}
		else if(buddy)
		{
			buddy.Die(owner, self, meansofdeath:"Betrayal");
			deployState = 0;
		}
		return false;
	}
	
	override void OnDrop(Actor dropper)
	{
		if(buddy)
			buddy.Die(owner, self, meansofdeath:"Betrayal");
		deployState = 0;
	}
	
	States
	{
	Spawn:
		FZB1 A -1;
		Stop;
	}
}
