class BDP_FreezerBot : BDP_Equipment
{
	Default
	{
		Tag "Freezer Bot";
		Inventory.Icon "FZB1A1";
		Inventory.PickupMessage "Picked up a freezer bot.";
		Inventory.PickupSound "freezebot/talk";
		BDP_Equipment.Cooldown 120;
		Scale 0.8;
	}
	
	FreezerBot buddy;
	int deployState;
	
	override void DoEffect()
	{
		if(deployState == 2 && (!buddy || buddy.health < 1))
		{
			deployState = 0;
			TriggerCooldown();
		}
		super.DoEffect();
	}
	
	override bool UseEffect()
	{
		if(deployState == 0)
		{
			owner.TakeInventory("BDP_FreezerBotStates", 1);
			owner.GiveInventory("BDP_FreezerBotStates", 1);
			deployState = 1;
		}
		else if(buddy)
		{
			buddy.Die(owner, self, meansofdeath:"Betrayal");
			deployState = 0;
		}
		return false;
	}
	
	override void OnDrop(Actor dropper)
	{
		if(buddy)
			buddy.Die(owner, self, meansofdeath:"Betrayal");
		deployState = 0;
	}
	
	States
	{
	Spawn:
		FZB1 A -1;
		Stop;
	}
}

class BDP_FreezerBotStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	override void Tick()
	{
		return;
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.canLedgeClimb = false;
			A_ClearOverlays(49,50);
			A_Overlay(90, "PlaceFreezerBot");
			player.SetPSprite(1, player.readyweapon.ResolveState("TossItem"));
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	PlaceFreezerBot:
		TNT1 A 5;
		FR2R A 1 {A_StartSound("freezebot/talk",7); A_StartSound("foley/ASGDrumOut",8);}
		FR2R BCDEFGHIJJJJJJJJJ 1;
		FR2R K 1
		{
			let buddy = FreezerBot(spawn("FreezerBot", (pos.x, pos.y, pos.z + 40)));
			let equip = BDP_FreezerBot(FindInventory("BDP_FreezerBot"));
			if(buddy)
			{
				buddy.angle = angle + 180;
				buddy.master = invoker.owner;
				if(equip)
				{
					equip.buddy = buddy;
					equip.deployState = 2;
				}
			}
		}
		FR2R LMNOPQRST 1;
		TNT1 A 5;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			TakeInventory("BDP_FreezerBotStates", 1);
		}
		Stop;
	}
}
