class BDP_RevenantLauncher : BDP_Equipment
{
	Default
	{
		Tag "Revenant Launchers";
		Inventory.Icon "FATBD0";
		Inventory.PickupMessage "Picked up the revenant launchers.";
		Inventory.PickupSound "revup";
		BDP_Equipment.Cooldown 45;
	}
	
	override bool UseEffect()
	{
		owner.TakeInventory("BDP_RevenantLauncherStates", 1);
		owner.GiveInventory("BDP_RevenantLauncherStates", 1);
		return true;
	}
	
	States
	{
	Spawn:
		FATB D -1;
		Stop;
	}
}

class BDP_RevenantLauncherStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	int burstCount;
	
	override void Tick()
	{
		return;
	}
	
	action void A_QuadSound()
	{	
		If(invoker.owner.findinventory("BDP_PowerQuadDamage"))
		{
			invoker.owner.a_startsound("items/quaddamage/active", 103);
		}
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			A_Overlay(90, "FireLaunchers");
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	FireLaunchers:
		RVCG JKLMN 2;
		RVCG A 5;
	BurstFire:
		RVCG B 1
		{
			A_FireProjectile("BDP_PlayerRevenantSeekerMissiles", spawnofs_xy:-20);
			A_StartSound("RLANSHOT", 69);
			A_StartSound("skeleton/attack", 70);
			A_QuadSound();
		}
		RVCG C 1;
		RVCG F 1
		{
			A_FireProjectile("BDP_PlayerRevenantSeekerMissiles", spawnofs_xy:20);
			A_StartSound("RLANSHOT", 69);
			A_StartSound("skeleton/attack", 70);
			A_QuadSound();
		}
		RVCG G 1;
		TNT1 A 0
		{
			if(invoker.burstCount < 5)
			{
				invoker.burstCount++;
				return resolvestate("BurstFire");
			}
			return resolvestate(null);
		}
		RVCG A 5;
		RVCG NMLKJ 2;
		TNT1 A 0 TakeInventory("BDP_RevenantLauncherStates", 1);
		Stop;
	}
}
