class BDP_LandMine : BDP_Equipment
{
	Default
	{
		Tag "Land Mine";
		Inventory.Icon "PIPBB0";
		Inventory.PickupMessage "Picked up a Land Mine.";
		Inventory.PickupSound "PIPESEL";
		BDP_Equipment.Cooldown 60;
		Scale 0.4;
	}
	
	override bool UseEffect()
	{
		//owner.A_StartSound("DukeThrw");
		//owner.SpawnPlayerMissile("BDP_LandMineTHrownQuickTossed");
		owner.TakeInventory("BDP_LandMineStates", 1);
		owner.GiveInventory("BDP_LandMineStates", 1);
		return true;
	}
	
	States
	{
	Spawn:
		PIPB B -1;
		Stop;
	}
}

class BDP_LandMineStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	override void Tick()
	{
		return;
	}
	
	action void A_QuadSound()
	{	
		If(invoker.owner.findinventory("BDP_PowerQuadDamage"))
		{
			invoker.owner.a_startsound("items/quaddamage/active", 103);
		}
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.canLedgeClimb = false;
			A_ClearOverlays(49,50);
			A_Overlay(90, "TossLandMine");
			player.SetPSprite(1, player.readyweapon.ResolveState("TossItem"));
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	TossLandMine:
		TNT1 A 5;
		TNT1 A 0 A_startsound("MineArm",2);
		DKPB PQRSTUFGHI 1;
		TNT1 A 0 
		{
			A_FireProjectile("BDP_LandMineThrown",0,0,0,0);
			A_StartSound("DukeThrw");
			A_QuadSound();
		}
		DKPB JKKKLLMNO 1;
		TNT1 A 5;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			TakeInventory("BDP_LandMineStates", 1);
		}
		Stop;
	}
}

CLASS BDP_LandMineTHrownIdle: Actor
{
	float beepvolume;
	int beeptics;
	Default
	{
	scale 0.4;
	radius 8;
	height 4;
	//+THRUACTORS 
	+FORCEXYBILLBOARD;
	 Species "ThrownPipebombs";
	 +flatsprite;
	+MOVEWITHSECTOR;
	//+shootable;
	-BLOODSPLATTER;
	//Health 100;
	//+DONTTHRUST;
	//+noblood;
	}
	 States
	{
	Spawn:
		TNT1 A 0 NoDelay A_startsound("PIPBOING", 8);
		TNT1 A 0
		{
			beepvolume = 1.0;
		}
	Beep:
		PIPb D 1 
		{
			A_startsound("minebep",CHAN_BODY,0,beepvolume);
			A_alertmonsters();
			beeptics = 0;
		}
	KeepSpawned:
		PIPb CCCCCCCCCCCCCCC 1 
		{
			bool scarymonster;
			int beeppause;
			beeptics++;
			double closestmonster = 9000;
			BlockThingsIterator CheckForTargets = BlockThingsIterator.Create(self,256);
			Actor CurrentActor; //A pointer to whatever actor the iterator is iterating through.
			While (CheckForTargets.Next()) 
			{
				 CurrentActor = CheckForTargets.Thing;
				//If the actor is a monster, has none of the specified item, the caller has a line of sight to the actor, and the actor is within 512 MU, then jump to the see state.
				//Itemname obviously has to be whatever item you want the actor to check that the possible target has none of, and the 512 map unit sight range can be changed to anything else.
				If(CurrentActor && CurrentActor.bIsMonster && currentactor.bshootable && !currentactor.bfriendly && !currentactor.findinventory("isplayer") && CheckSight(CurrentActor,SF_IGNOREWATERBOUNDARY) && currentactor.health > 0)
				{
					
					
					If((distance3d(currentactor) - currentactor.radius) < closestmonster)
					{
						closestmonster = distance3d(currentactor) - currentactor.radius;
					}
					
					If(distance3d(currentactor) < ( 30 + currentactor.radius))
					{
						scarymonster = true;
						If(currentactor.findstate("pain"))
						{
							currentactor.setstatelabel("pain.mine");
						}
					}
				}
			}
			//console.printf("%d",closestmonster);
			
			If(beepvolume > 0.3 && closestmonster > 180)
			{
				Beepvolume -= 0.001;
				BeepPause = 35;
			}
			Else If(closestmonster < 180)
			{
				Beepvolume = 1.0;
				Beeppause = 30;
				If(closestmonster > 165 && closestmonster < 180)
				{
					beeppause = 27;
				}
				Else If(closestmonster > 150)
				{
					beeppause = 24;
				}
				Else If(closestmonster > 135)
				{
					beeppause = 21;
				}
				Else If(closestmonster > 120)
				{
					beeppause = 18;
				}
				Else If(closestmonster > 105)
				{
					beeppause = 15;
				}
				Else If(closestmonster > 90)
				{
					beeppause = 12;
				}
				Else If(closestmonster > 75)
				{
					beeppause = 9;
				}
				Else If(closestmonster > 60)
				{
					beeppause = 6;
				}
				Else If(closestmonster < 60)
				{
					beeppause = 4;
				}
			}
			Else
			{
				beeppause = 35;
			}
			If(scarymonster)
			{
				return ResolveState("MakeItExplode");
			}
			If(beeptics >= beeppause)
			{
				return resolvestate("beep");
			}
			return ResolveState(null);
		}
		Loop;
	MakeItExplode:
		TNT1 AAA 0;
		NULL A 0 A_StartSound("MINEDING",CHAN_BODY,0,1.0);
		NULL A 0 A_StartSound("MINEBOOP",7,0,0.7);
		PIPb D 20;
		Death:
		NULL A 0 A_StopSound();
		NULL A 0 A_StopSound(7);
	Explosion:
		TNT1 A 0
		{
			A_StartSound("PIPEEXPLODE", 0);
			A_StartSound("FAREXPL", 0);
			A_AlertMonsters();
			A_Explode(640, 256, fulldamagedistance:64, damagetype:"Explosive");
			A_SpawnItemEx ("Footstep91", zofs:40);
			A_QuakeEx(9, 9, 9, 8, 0, 960);
			A_SpawnItemEx("BDP_RicochetExplosion");
			A_SpawnItemEx("DetectCeilCrater");
			A_SpawnItemEx("UnderwaterExplosion");
			A_SpawnItemEx("ExplosionFlareSpawner");
		}
		TNT1 AAAAAAAAAAAAAA 0 A_SpawnProjectile ("ExplosionParticleHeavy", random(0, 64), random(-64, 64), random (0, 360), 2, random (0, -180));
		TNT1 AAAAAA 0 A_SpawnProjectile ("ExplosionParticleHeavy", random(0, 64), random(-64, 64), random (0, 180), 2, random (0, 180));
		TNT1 AA 0 A_CustomMissile ("SpawnedExplosion", random(3,-3), random(6,-6), random (0, 360), 2, random (0, 360));
	   	TNT1 AAAA 0 A_spawnProjectile ("SmallExplosionFlames", random(0, 64), random(-64, 64), random (0, 360), 2, random (0, 360));
		TNT1 AAAAAAA 0 A_SpawnProjectile ("ExplosionFlames", random(0, 64), random(-64, 64), random (0, 360), 2, random (0, 360));
		TNT1 AAAA 0 A_CustomMissile ("MediumExplosionFlames", random(0, 64), random(-64, 64), random (0, 360), 2, random (0, 360));
		TNT1 AAAA 0 A_SpawnItemEx ("PipebombPiece2",random(5,-5),random(6,1),random(5,2),random(14,3),random(15,6),random(3,20),random(360,1),SXF_NOCHECKPOSITION,0);
		TNT1 AAA 0 A_SpawnItemEx ("PipebombPiece",random(5,-5),random(6,1),random(5,2),random(14,3),random(15,6),random(3,20),random(360,1),SXF_NOCHECKPOSITION,0);
		TNT1 AAAAAAAAAAAAA 0 A_CustomMissile ("BDExplosionparticlesBig", random(-4, 4), random(-5,5), random (0, 360), 2, random (0, 360));
		TNT1 AAAAAAAA 0 A_CustomMissile ("BDExplosionparticles2", random(-4, 4), random(-5,5), random (0, 360), 2, random (0, 90));
		stop;
	}
}
