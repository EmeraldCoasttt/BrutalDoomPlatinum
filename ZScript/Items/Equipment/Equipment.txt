class BDP_Equipment : Inventory abstract
{
	Default
	{
		Inventory.MaxAmount 1;
		BDP_Equipment.Cooldown 0;
		BDP_Equipment.Charges 1;
		+INVENTORY.AUTOACTIVATE
		+BDP_EQUIPMENT.ACTIVE
	}
	
	int EquipFlags;
		
	double cooldown;
	int effectTimer;
	bool pickupDepleted;
	bool holding;
	
	flagdef CooldownNotTimer : EquipFlags, 0;
	flagdef Active : EquipFlags, 1;
	flagdef NoIcon : EquipFlags, 2;
	
	double maxCooldown;
	property Cooldown : maxCooldown;
	
	int charges;
	property Charges : charges;
	
	void TriggerCooldown(bool force = false)
	{
		int mod = bCooldownNotTimer ? 1 : TICRATE;
		if(force || !cooldown && bActive)
			cooldown = maxCooldown * mod;
		if(charges > 0) charges--;
	}
	
	override bool Use(bool pickup)
	{
		if(pickup)
		{
			PickupEquipment(pickupDepleted);
			pickupDepleted = true;
			return false;
		}
		if(charges)
		{
			bool cd = UseEffect();
			if(!bActive) return false;
			holding = true;
			if(cd)
			{
				TriggerCooldown();
			}
		}
		return false;
	}
	
	override void Tick()
	{
		if(!bCooldownNotTimer && cooldown > 0 && charges < default.charges) cooldown--;
		if(!cooldown && charges < default.charges)
		{
			charges++;
			if(charges < default.charges && !bCooldownNotTimer)
				cooldown = maxCooldown * TICRATE;
		}
		super.Tick();
	}
	
	override void DoEffect()
	{
		if(holding)
		{
			bool cancel = HoldEffect();
			if(cancel)
				ReleaseEquipment();
		}
	}
	
	void ReleaseEquipment()
	{
		if(holding && charges && bActive)
		{
			bool cd = ReleaseEffect();
			if(cd)
			{
				TriggerCooldown();
			}
		}
		holding = false;
	}
	
	virtual void PickupEquipment(bool depleted)
	{
		return;
	}
	
	virtual bool UseEffect()
	{
		return false;
	}
	
	virtual bool HoldEffect()
	{
		return false;
	}
	
	virtual bool ReleaseEffect()
	{
		return false;
	}
}
