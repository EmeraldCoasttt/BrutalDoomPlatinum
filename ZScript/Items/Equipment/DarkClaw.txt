class BDP_DarkClaw : BDP_Equipment
{
	Default
	{
		Tag "Dark Claw";
		Inventory.Icon "DCLWA0";
		Inventory.PickupMessage "Picked up the dark claw.";
		Inventory.PickupSound "GRNPIN";
		BDP_Equipment.Cooldown 2000;
		BDP_Equipment.Charges 1;
		+BDP_EQUIPMENT.COOLDOWNNOTTIMER
		+BDP_EQUIPMENT.REPLACESMELEE
	}
	
	override bool UseEffect()
	{
		owner.TakeInventory("BDP_DarkClawStates", 1);
		owner.GiveInventory("BDP_DarkClawStates", 1);
		let stContainer = BDP_DarkClawStates(owner.FindInventory("BDP_DarkClawStates"));
		if(!stContainer) return false;
		stContainer.forceActivate = true;
		return true;
	}
	
	override void QuickMelee()
	{
		owner.TakeInventory("BDP_DarkClawStates", 1);
		owner.GiveInventory("BDP_DarkClawStates", 1);
		let stContainer = BDP_DarkClawStates(owner.FindInventory("BDP_DarkClawStates"));
		if(!stContainer) return;
		stContainer.onCooldown = !(charges);
	}
	
	States
	{
	Spawn:
		DCLW A -1;
		Stop;
	}
}

class BDP_DarkClawStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	bool onCooldown;
	bool forceActivate;
	
	override void Tick()
	{
		return;
	}
	
	action void A_QuadSound()
	{	
		If(invoker.owner.findinventory("BDP_PowerQuadDamage"))
		{
			invoker.owner.a_startsound("items/quaddamage/active", 103);
		}
	}
	
	action void A_BDPMeleeStart(double range = 200)
	{
	//if(A_CheckMeleeRange(skipWalls:true) < range) A_Recoil3D(-20); 
		FLineTraceData lt;
			LineTrace(angle, range, pitch, 0, player.viewheight, data:lt);
			If(lt.hitactor && lt.hitactor.bsolid && lt.hitactor.bismonster)
			{
				A_face(lt.hitactor,0,0,0,0,FAF_MIDDLE);
				A_Recoil3D(-20);
			}
	}
	
	action void A_Recoil3D(double amt)
	{
		vel += BDP_Math.VecFromAngles(angle, pitch, -amt);
	}
	
	//from project brutality
	action void A_UseLine(int range = 64) {
		FLineTraceData useline;
		LineTrace(angle, range, pitch, offsetz:player.viewheight, data:useline);
		if (useline.HitLine && useline.LineSide == line.front)
		{
			bool isDoor = false;
			switch (useline.HitLine.special)
			{
			case Generic_Door:
			case Door_LockedRaise:
			case Door_Animated:
			case Door_Open:
			case Door_Raise:
				isDoor = true;
				break;
			}
			if(isDoor) return;
			useline.HitLine.Activate(player.mo, useline.LineSide, SPAC_Use);
		}
	}
	
	action Actor A_BDPMelee(double range = 200, name projectile = "MeleeAttack", double spawnheight = -7, bool doHitThrust = true)
	{
		Actor Victim;
		
		FLineTraceData lt;
		double aimz = self.player ? self.player.viewheight : (self.height * 0.5);
		self.LineTrace(self.angle, range, self.pitch, 0, aimz, data:lt);
		victim = lt.HitActor;
		int aimCheck = -6;
		bool backsmack;
		
		
		while (aimCheck++ < 6 && !victim)
		{
			self.LineTrace(self.angle + (aimCheck * 8), range, self.pitch, 0, aimz, data:lt);
			victim = lt.hitActor;
		}
			
		if(victim && victim.bSHOOTABLE && victim.bISMONSTER) 
		{
			vector3 victimAngles = level.sphericalCoords(victim.pos, self.pos, (victim.angle, victim.pitch));
			
			double inertia = BDP_Math.GetInertia(victim.mass);
			self.A_Quake(2,3,0,20,"");
			self.A_face(victim);
			if(doHitThrust && victim.bsolid)
			{
				self.vel *= 0;
				self.vel += BDP_Math.VecFromAngles(self.angle, self.pitch, -12);
			}
		}
		actor proj = A_FireProjectile(projectile, 0, 0, 0, spawnheight);
		If(proj && victim && victim.bSHOOTABLE)
		{
			proj.setorigin(victim.pos,false);
		}
		
		return victim;
	}
	
	action void A_SpawnBonuses(actor victim, int damage, int amount = 1)
	{
		if(victim.health - victim.ApplyDamageFactor("Fire", damage) < 1)
		{
			class<Actor> bonus = "HealthPlusFatalityBonus";
			if(victim.FindInventory("firetic") || victim is "BDP_LostSoul")
				bonus = "ArmorShardFatalityBonus";
			for(int i = 0; i < amount; i++)
				victim.A_SpawnItemEx(bonus, xvel:frandom(-2.0, 2.0), frandom(-2.0, 2.0), frandom(4.0, 6.0));
		}
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.canLedgeClimb = false;
			A_ClearOverlays(49,50);
			A_Overlay(90, "SwingClaw");
			player.SetPSprite(1, player.readyweapon.ResolveState("DoingMelee"));
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	SwingClaw:
		TNT1 A 1;
		CLAW A 1;
		TNT1 A 0
		{
			A_StartSound("skeleton/swing", 0);
			A_StartSound("fstatk", 0);
		}
		CLAW BC 1 A_BDPMeleeStart();
		TNT1 A 0
		{
			Actor victim = A_BDPMelee(projectile:"MeleeAttackDarkClaw");
			if(victim && victim is "BDP_Headshot") victim = victim.tracer;
			if(!invoker.forceActivate) A_UseLine();
			A_QuadSound();
			if((victim && victim.bISMONSTER && !victim.bFRIENDLY || invoker.forceActivate) && !invoker.onCooldown)
			{
				A_StartSound("items/darkclaw/blast", 0);
				let dplr = BDP_PlayerPawn(invoker.owner);
				if(dplr && dplr.ownedEquipment is "BDP_DarkClaw") dplr.ownedEquipment.TriggerCooldown();
				BlockThingsIterator it = BlockThingsIterator.Create(self, 200);
				
				while(it.Next())
				{
					target = it.thing;
					if(target && target.bISMONSTER && !target.bFRIENDLY && (target.health > 0 || target == victim) && CheckLOF(CLOFF_SKIPENEMY|CLOFF_SKIPFRIEND|CLOFF_SKIPOBJECT, ptr_target:AAPTR_TARGET))
					{
						double distance = Distance3D(target);
						if(distance > 200) continue;
						int damage = 500 * ((200 - distance) / 200);
						A_SpawnBonuses(target, damage, 3);
						target.DamageMobj(invoker, invoker.owner, damage, "Fire");
					}
				}
				
				for(int i = 0; i < 36; i++)
				{
					Vector2 fvel = AngleToVector(angle + i * 10, 16);
					A_SpawnItemEx("SmallExplosionFlames", zofs:32, fvel.x, fvel.y);
				}
			}
		}
		CLAW DEFGH 1;
		TNT1 A 5;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			let dplr = BDP_PlayerPawn(invoker.owner);
			if(!dplr) return;
			dplr.blockKick = true;
			TakeInventory("BDP_DarkClawStates", 1);
		}
		Stop;
	}
}

class MeleeAttackDarkClaw : MeleeAttack
{
	Default
	{
		Damagetype "cut";
		MeleeAttack.BaseDamage 66;
		decal "SawHorizontal";
		+BLOODSPLATTER;
	}
	
	override int DoSpecialDamage(Actor victim, int damage, Name damagetype)
	{
		if(victim.health - victim.ApplyDamageFactor(damagetype, damage) < 1)
		{
			class<Actor> bonus = "HealthPlusFatalityBonus";
			if(victim.FindInventory("firetic") || victim is "BDP_LostSoul")
				bonus = "ArmorShardFatalityBonus";
			victim.A_SpawnItemEx(bonus, xvel:frandom(-2.0, 2.0), frandom(-2.0, 2.0), frandom(4.0, 6.0));
		}
		return super.DoSpecialDamage(victim, damage, damagetype);
	}
	
	States
	{
	Spawn:
        TNT1 A 1;
		Stop;
	Death:
		TNT1 A 0;
		TNT1 A 0
		{
			if(target.findinventory("sequencebreaker"))
				BDP_DoorBuster.DestroyDoor(target,  breakLocks: BDP_DoorBuster.LB_All);
			else
				BDP_DoorBuster.DestroyDoor(target,  breakLocks: BDP_DoorBuster.LB_CheckKey);
			A_SpawnItemEx("BDP_Ricochet",0,0,0,0,0,0,180);
			A_StartSound("baron/melee", 0);
			Radius_Quake(2, 3, 0, 4, 0);
		}
		Stop;
	XDeath:
		TNT1 A 0
		{
			Radius_Quake(2, 3, 0, 4, 0);
			A_StartSound("CSAWFLESH", 0);
			A_StartSound("baron/melee", 0);
		}
		Stop;
	}
