class BDP_FragGrenade : BDP_Equipment
{
	Default
	{
		Tag "Frag Grenade";
		Inventory.Icon "GRNDA";
		Inventory.PickupMessage "Picked up a Frag Grenade.";
		Inventory.PickupSound "GRNPIN";
		BDP_Equipment.Cooldown 20;
		BDP_Equipment.Charges 2;
		Scale 0.5;
	}
	
	override bool UseEffect()
	{
		owner.TakeInventory("BDP_FragGrenadeStates", 1);
		owner.GiveInventory("BDP_FragGrenadeStates", 1);
		return true;
	}
	
	States
	{
	Spawn:
		GRND D -1;
		Stop;
	}
}

class BDP_FragGrenadeStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	int fuse;
	const MAX_FUSE = 48;
	
	override void Tick()
	{
		return;
	}
	
	action void A_QuadSound()
	{	
		If(invoker.owner.findinventory("BDP_PowerQuadDamage"))
		{
			invoker.owner.a_startsound("items/quaddamage/active", 103);
		}
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.canLedgeClimb = false;
			A_ClearOverlays(49,50);
			A_Overlay(90, "TossGrenade");
			player.SetPSprite(1, player.readyweapon.ResolveState("TossItem"));
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	TossGrenade:
		TNT1 A 5;
		GRTH ABCDEF 1;
		TNT1 A 0 A_StartSound ("GRNPIN");
		GRTH FGHIJK 1;
	Cook:
		TNT1 A 2;
		TNT1 A 0
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			if(dplr.ownedEquipment.holding && invoker.fuse < MAX_FUSE)
			{
				invoker.fuse++;
				if(invoker.fuse % 8 == 0) A_StartSound("BEP");
				return resolvestate("Cook");
			}
			if(invoker.fuse == MAX_FUSE)
				return resolvestate("KillYourself");
			return resolvestate(null);
		}
		TNT1 A 0 A_StartSound ("CS16uwu");
		TNT1 A 0 A_Jump(2,2);
		TNT1 A 0 A_StartSound ("GRNTOSS");
		GRTH LM 1;
		TNT1 A 0
		{
			let grn = BDP_FragGrenadeThrown(A_FireProjectile("BDP_FragGrenadeThrown"));
			if(grn) grn.fuse = invoker.fuse;
			A_QuadSound();
		}
		GRTH NOP 1;
		TNT1 A 5;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			TakeInventory("BDP_FragGrenadeStates", 1);
		}
		Stop;
	KillYourself:
		TNT1 A 0
		{
			let grn = BDP_FragGrenadeThrown(A_FireProjectile("BDP_FragGrenadeThrown"));
			if(grn) grn.fuse = invoker.fuse;
			A_QuadSound();
		}
		TNT1 A 30;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			TakeInventory("BDP_FragGrenadeStates", 1);
		}
		Stop;
	}
}

class BDP_FragGrenadeThrown : Actor
{
	Default
	{
		Scale 0.5;
		Radius 4;
		Height 4;
		Projectile;
		Speed 30;
		DamageFunction 2;
		Gravity 0.7;
		Projectile;
		BounceFactor 0.60;
		WallBounceFactor 0.40;
		BounceSound "HandNadeBounce";
		Obituary "%o ate %k grenade.";
		+BOUNCEONFLOORS
		+BOUNCEONWALLS
		+BOUNCEONCEILINGS
		+ALLOWBOUNCEONACTORS
		+BOUNCEAUTOOFFFLOORONLY
		+CANBOUNCEWATER
		+SKYEXPLODE
		+FORCEXYBILLBOARD
		+ROLLSPRITE
		+ROLLCENTER
		+MTHRUSPECIES
		+XFLIP
		-NOTELEPORT
		-NOGRAVITY
	}
	
	const FUSE_MAX = 48;
	int fuse;
	
	override void Tick()
	{
		fuse++;
		Super.Tick();
	}
	
	States
	{
	Spawn:
		GRND A 1 NoDelay
		{
			A_SetRoll(roll + 22.5);
			if(fuse > FUSE_MAX) return findstate("Explode");
			return findstate(null);
		}
		Loop;
	Death:
		TNT1 A 0
		{
			A_SetRoll(45);
			A_StopSound();
			A_StartSound("HandNadeBounce", 0);
			A_AlertMonsters(128);
			A_SpawnItem("GrenadeAlerter");
		}
	DeathLoop:
		GRND A 1
		{
			A_CustomMissile("GrenadeSmoke", 0, 0, random (-6,6), 2, random (0, 360));
			if(fuse > FUSE_MAX) return findstate("Explode");
			return findstate(null);
		}
		Loop;
	Explode:
		TNT1 A 0 A_NoBlocking();
		TNT1 A 0 A_SpawnItem("FragGrenadeExplosion");
		TNT1 A 0 A_StopSound();
		TNT1 A 0 A_PlaySound("GRENEXPL");
		Stop;
	}
}

class HandGrenade : BDP_FragGrenadeThrown {}
