class BDP_Tether : BDP_Equipment
{
	Default
	{
		Tag "Dimensional Tether";
		Inventory.PickupMessage "Picked up a dimensional tether!";
		Inventory.PickupSound "items/computermap";
		BDP_Equipment.Cooldown 15;
		Inventory.Icon "TETHA0";
		Scale 0.5;
	}
	
	BDP_TetherDest linkPoint;
	
	override void DoEffect()
	{
		if(linkPoint)
		{
			if(linkPoint.targetInRange)
			{
				if(icon == TexMan.CheckForTexture("TETHB0"))
					owner.A_StartSound("BEPBEP");
				icon = TexMan.CheckForTexture("TETHC0");
			}
			else
			{
				icon = TexMan.CheckForTexture("TETHB0");
			}
		}
		super.DoEffect();
	}
	
	override bool UseEffect()
	{
		if(!linkPoint)
		{
			owner.A_StartSound("BEPBEP", 0);
			linkPoint = BDP_TetherDest(Spawn("BDP_TetherDest", owner.pos));
			icon = TexMan.CheckForTexture("TETHB0");
			return false;
		}
		else
		{
			owner.A_SpawnItemEx("TeleportFog");
			linkPoint.A_SpawnItemEx("TeleportFog");
			linkPoint.A_StopSound();
			owner.TeleportMove(linkPoint.pos, true);
			linkPoint.Destroy();
			owner.A_StartSound("Weapons/Unmaker/UseTeleporter", 0);
			icon = TexMan.CheckForTexture("TETHA0");
			return true;
		}
	}
	
	override void OnDrop(Actor dropper)
	{
		if(linkPoint)
		{
			linkPoint.Destroy();
			linkPoint = null;
		}
	}
	
	States
	{
	Spawn:
		TETH A -1;
		Stop;
	}
}

class BDP_TetherDest : Actor
{
	Default
	{
		-SOLID
		+BLOCKEDBYSOLIDACTORS
		+DONTTHRUST
		Radius 16;
		Height 56;
		Scale 0.01;
		+FORCEXYBILLBOARD
		+SQUAREPIXELS
		+rollsprite
		+rollcenter
	}
	
	bool targetInRange;
	
	override void Tick()
	{
		super.Tick();
		BlockThingsIterator it = BlockThingsIterator.Create(self, 1);
		while (it.Next())
		{
			if(!it.thing || !it.thing.bISMONSTER || CheckPosition(pos.xy, true)) continue;
			if(!targetInRange) A_StartSound("BEPBEP");
			targetInRange = true;
			return;
		}
		targetInRange = false;
	}
	
	States
	{
	Spawn:
		TETH DEF 1 BRIGHT 
		{
			A_StartSound("TETHHUM", flags:CHANF_LOOPING);
			A_SetRoll(roll - 69);
			A_SpawnItemEx("TetherSpark", random(10, -10), random(10, -10), random(30,50), random(-0.8, 0.8), random(-0.8, 0.8), random(-0.5, 0.5), random(-1,-360), SXF_CLIENTSIDE|SXF_NOCHECKPOSITION);
			If(scale.x < 0.5)
			{
				scale.x += 0.04;
				scale.y += 0.04;
			}
		}
		LOOP;
	}
}
