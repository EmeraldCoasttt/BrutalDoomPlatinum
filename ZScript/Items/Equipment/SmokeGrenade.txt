class BDP_SmokeGrenade : BDP_Equipment
{
	Default
	{
		Tag "Thermite Grenade";
		Inventory.Icon "GRNDB0";
		Inventory.PickupMessage "Picked up a thermite grenade.";
		Inventory.PickupSound "GRNPIN";
		BDP_Equipment.Cooldown 30;
		Scale 0.75;
	}
	
	override bool UseEffect()
	{
		owner.TakeInventory("BDP_SmokeGrenadeStates", 1);
		owner.GiveInventory("BDP_SmokeGrenadeStates", 1);
		return true;
	}
	
	States
	{
	Spawn:
		GRND B -1;
		Stop;
	}
}

class BDP_SmokeGrenadeStates : CustomInventory
{
	Default
	{
		Inventory.MaxAmount 1;
		+INVENTORY.AUTOACTIVATE
	}
	
	override void Tick()
	{
		return;
	}
	
	action void A_QuadSound()
	{	
		If(invoker.owner.findinventory("BDP_PowerQuadDamage"))
		{
			invoker.owner.a_startsound("items/quaddamage/active", 103);
		}
	}
	
	States
	{
	Use:
		TNT1 A -1
		{
			let dplr = BDP_PlayerPawn(invoker.owner);
			dplr.canLedgeClimb = false;
			A_ClearOverlays(49,50);
			A_Overlay(90, "TossGrenade");
			player.SetPSprite(1, player.readyweapon.ResolveState("TossItem"));
			A_OverlayOffset(90); //no args resets it to default position.
		}
		Wait;
	TossGrenade:
		TNT1 A 5;
		3RTH ABCDE 1;
		TNT1 A 0 A_StartSound("ICEGPRE2", 0);
		3RTH F 3;
		3RTH GHIJK 1;
		TNT1 A 2;
		TNT1 A 0 A_StartSound ("CS16uwu");
		TNT1 A 0 A_Jump(2,2);
		TNT1 A 0 A_StartSound ("GRNTOSS");
		GRTH OP 1;
		TNT1 A 0
		{
			A_FireProjectile("SmokeSpewer");
			A_QuadSound();
		}
		GRTH Q 1;
		GRTH RSTU 1;
		TNT1 A 5;
		TNT1 A 0
		{
			player.SetPSprite(1, player.readyweapon.ResolveState("FinishToss"));
			TakeInventory("BDP_SmokeGrenadeStates", 1);
		}
		Stop;
	}
}

class SmokeSpewer : Actor
{
	Default
	{
		Radius 4;
		Height 4;
		Speed 30;
		Gravity 0.7;
		DamageFunction 20;
		DamageType "Kick";
		Projectile;
		Scale 0.5;
		BounceFactor 0.25;
		WallBounceFactor 0.05;
		BounceCount 2;
		BounceSound "Weapons/GrenadeBounce";
		DeathSound "Weapons/GrenadeBounce";
		+BOUNCEONWALLS
		+BOUNCEONFLOORS
		+BOUNCEONCEILINGS
		+ALLOWBOUNCEONACTORS
		+BOUNCEONACTORS
		+CANBOUNCEWATER
		+FORCEXYBILLBOARD
		+ROLLSPRITE
		+ROLLCENTER
		-NOGRAVITY
	}
	States
	{
	Spawn:
		GRND B 1 A_SetRoll(roll + 22.5);
		Loop;
	Death:
		GRND E 2
		{
			A_SetRoll(0);
			A_SpawnProjectile("BlockingSmoke", 0, 0, frandom(0, 360), CMF_AIMDIRECTION, frandom(0, -75));
			A_FadeOut(0.004);
		}
		Loop;
	}
}
