class BDP_Backpack : BackpackItem replaces Backpack
{
	Default
	{
		Height 26;
		Inventory.PickupMessage "$GOTBACKPACK";
		Inventory.PickupSound "BACKPACK";
	}
	States
	{
	Spawn:
		TNT1 A 0 NoDelay A_Jump(64, "SpawnOriginal");
		TNT1 A 0 A_Jump(256, "Spawn1", "Spawn2", "Spawn3");
		Loop;
	SpawnOriginal:
		BPAK A -1;
	Spawn1:
		BPAK BCDE 4;
		LOOP;
	Spawn2:
		BPAK CDEB 4;
		LOOP;
	Spawn3:
		BPAK DEBC 4;
		LOOP;
	}
	override bool TryPickup (in out Actor toucher)
	{
		if(super.trypickup(Toucher))
		{
			toucher.Score += 500;
			return true;
		}
		else
		{
			return false;
		}
	}
}

class BDP_Ammo : Ammo
{
	Default
	{
		BDP_Ammo.ScoreAmount 50;
		+FORCEXYBILLBOARD;
	}
	
	mixin BDP_ScoreMixin;
	
	override Class<Ammo> GetParentAmmo()
	{
		class<Object> type = GetClass();
		
		while(type.GetParentClass() != "BDP_Ammo" && type.GetParentClass() != null)
		{
			type = type.GetParentClass();
		}
		return (class<Ammo>)(type);
	}
}

class BDP_MultiAmmo : Inventory
{
	string ammotype1;
	property ammotype1 : ammotype1;
	string ammotype2;
	property ammotype2 : ammotype2;
	int ammoamount1;
	property ammoamount1 : ammoamount1;
	int ammoamount2;
	property ammoamount2 : ammoamount2;
	
	Default
	{
		+ForceXyBillboard;
	}
	
	override bool TryPickup(in out Actor toucher)
	{
	  bool success;
	  Ammo a1 = Ammo(Actor.Spawn(ammotype1, pos));
	  Ammo a2 = Ammo(Actor.Spawn(ammotype2, pos));
	  if (a1)
	  {
		a1.amount = ammoamount1;
		if (a1.CallTryPickup(toucher))
		  success = true;
		else
		  a1.Destroy();
	  }
	  if (a2)
	  {
		a2.amount = ammoamount2;
		if (a2.CallTryPickup(toucher))
		  success = true;
		else
		  a2.Destroy();
	  }
	  if (success) GoAwayAndDie();
		return success;
	  
	}

}
